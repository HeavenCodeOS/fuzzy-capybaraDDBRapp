pragma solidity ^0.8.0;

import "@openzeppelin/contracts/token/ERC721/extensions/ERC721URIStorage.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract FARAHhamsaSBT is ERC721URIStorage, Ownable {
    uint256 public tokenIdCounter;

    constructor() ERC721("FARAHhamsa™", "FARAH33925") {}

    function mintSoulbound(address to, string memory tokenURI) public onlyOwner {
        uint256 newId = tokenIdCounter;
        _safeMint(to, newId);
        _setTokenURI(newId, tokenURI);
        tokenIdCounter++;
    }

    function _beforeTokenTransfer(address from, address to, uint256 tokenId) internal pure {
        require(from == address(0), "Soulbound Token: Non-transferable");
    }
}
